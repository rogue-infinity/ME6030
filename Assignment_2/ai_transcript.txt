\section*{Theory Question 1}

Let $P = A(A\T A)^{-1}A\T$ be the projection matrix onto the
column space of $A$, where $A \in \R^{m \times n}$ has
linearly independent columns.

\subsection*{(a) $P$ is symmetric: $P\T = P$}

\begin{proof}
Taking the transpose and using $(XY)\T = Y\T X\T$:
\[
  P\T = \bigl(A(A\T A)^{-1}A\T\bigr)\T
      = (A\T)\T \bigl((A\T A)^{-1}\bigr)\T A\T
      = A \bigl((A\T A)\T\bigr)^{-1} A\T.
\]
Since $(A\T A)\T = A\T A$, we get $P\T = A(A\T A)^{-1}A\T = P$.
\end{proof}

\subsection*{(b) $P$ is idempotent: $P^2 = P$}

\begin{proof}
\[
  P^2 = A(A\T A)^{-1}A\T \cdot A(A\T A)^{-1}A\T
      = A \underbrace{(A\T A)^{-1}(A\T A)}_{= I}
        (A\T A)^{-1}A\T
      = A(A\T A)^{-1}A\T = P.
\]
\end{proof}

\subsection*{(c) $I - P$ is a projection matrix}

\begin{proof}
We verify the two properties of a projection matrix.

\textit{Symmetry:} $(I-P)\T = I\T - P\T = I - P$,
using part (a).

\textit{Idempotency:}
\[
  (I-P)^2 = I - 2P + P^2 = I - 2P + P = I - P,
\]
using part (b).

Hence $I-P$ is a projection matrix. It projects onto the
\emph{left null space} of $A$, i.e., $\mathrm{null}(A\T)$.
For any $v$, $(I-P)v = v - Pv$, and $Pv \in \mathrm{col}(A)$,
so $(I-P)v \in \mathrm{col}(A)^\perp = \mathrm{null}(A\T)$.
\end{proof}

\subsection*{(d) Eigenvalues of $P$ are only 0 and 1}

\begin{proof}
Let $Pv = \lambda v$ for some eigenvector $v \neq 0$.
Apply $P$ to both sides:
\[
  P^2 v = \lambda P v = \lambda^2 v.
\]
From part (b), $P^2 v = Pv = \lambda v$, so
\[
  \lambda^2 = \lambda \implies \lambda(\lambda-1) = 0
  \implies \lambda \in \{0,1\}.
\]
\end{proof}

\section*{Theory Question 2}

Given an inconsistent system $A\mathbf{x} = \mathbf{b}$
where $A \in \R^{m \times n}$, $m > n$.

\subsection*{(a) Normal equations}

\begin{proof}
Minimise $f(\mathbf{x}) = \|A\mathbf{x}-\mathbf{b}\|^2$.
Expanding and differentiating:
\[
  \nabla_{\mathbf{x}} f = 2A\T A\mathbf{x} - 2A\T\mathbf{b} = 0
  \implies A\T A\hat{x} = A\T\mathbf{b}.
\]
\end{proof}

\subsection*{(b) Residual orthogonality}

\begin{proof}
Let $\mathbf{r} = \mathbf{b} - A\hat{x}$. From the normal
equations:
\[
  A\T\mathbf{r} = A\T\mathbf{b} - A\T A\hat{x} = 0,
\]
so $\mathbf{r} \perp \mathrm{col}(A)$.
\end{proof}

\subsection*{(c) $A\T A$ is symmetric positive definite}

\begin{proof}
\textit{Symmetry:} $(A\T A)\T = A\T A$.

\textit{Positive definiteness:} For $\mathbf{x} \neq 0$:
\[
  \mathbf{x}\T(A\T A)\mathbf{x} = \|A\mathbf{x}\|^2 > 0,
\]
since $A$ full column rank implies $A\mathbf{x} \neq 0$.
Hence $A\T A$ is invertible and the solution
$\hat{x} = (A\T A)^{-1}A\T\mathbf{b}$ is unique.
\end{proof}
